function getSheet_() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sh = ss.getSheetByName(SHEET_NAME);
  if (!sh) { sh = ss.insertSheet(SHEET_NAME); }
  if (sh.getLastRow() === 0) { sh.appendRow(HEADERS); }
  return sh;
}

function jsonOut_(obj) {
  return ContentService
    .createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

// GET: a teljes ranglista JSON tömbként (soronként egy gép objektuma).
function doGet() {
  const sh = getSheet_();
  const values = sh.getDataRange().getValues();
  const rows = [];
  for (let i = 1; i < values.length; i++) {
    const r = {};
    HEADERS.forEach((h, j) => { r[h] = values[i][j]; });
    rows.push(r);
  }
  return jsonOut_(rows);
}

// POST: kétféle adat érkezhet - naplófájl vagy benchmark-eredmény.
function doPost(e) {
  let data;
  try {
    data = JSON.parse(e.postData.contents);
  } catch (err) {
    return jsonOut_({ ok: false, error: 'Hibás JSON: ' + String(err) });
  }

  // 1) NAPLÓ-ÁGAK. Egyik sem nyúl a táblázathoz, ezért nem kérnek zárat.
  if (data && data.action === 'log') {
    return saveDriverLog_(data);          // feltöltés - jelszó nélkül (lásd a konstansnál)
  }
  if (data && (data.action === 'loglist' || data.action === 'logget')) {
    if (!checkLogPassword_(data.pw)) {
      // Szándékos késleltetés: enélkül a rövid jelszó másodpercek alatt végigpróbálható.
      Utilities.sleep(2000);
      return jsonOut_({ ok: false, error: 'Hibás jelszó.' });
    }
    return data.action === 'loglist' ? listDriverLogs_() : getDriverLog_(data);
  }

  // 2) BENCHMARK-ÁG: egy gép eredményének beszúrása / frissítése (upsert a machine_id-re).
  const lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    const sh = getSheet_();
    const values = sh.getDataRange().getValues();
    let rowIndex = -1;
    for (let i = 1; i < values.length; i++) {
      if (String(values[i][0]) === String(data.machine_id)) { rowIndex = i + 1; break; }
    }
    const row = HEADERS.map(h => (data[h] !== undefined && data[h] !== null) ? data[h] : '');
    if (rowIndex > 0) {
      sh.getRange(rowIndex, 1, 1, HEADERS.length).setValues([row]);
    } else {
      sh.appendRow(row);
    }
    return jsonOut_({ ok: true });
  } catch (err) {
    return jsonOut_({ ok: false, error: String(err) });
  } finally {
    lock.releaseLock();
  }
}

/* A DriverVarázsló által küldött (gzip-elt, base64-es) naplófájl mentése a Drive-ra.
   A program a lánc legvégén küldi, tehát a fájlban benne van a záró jelentés és a
   teljes futásidő is. A fájlnév: <időbélyeg>_<gépnév>_build<szám>.log.gz
   Megnyitás: töltsd le és bontsd ki (7-Zip / WinRAR), sima szövegfájl lesz. */
function saveDriverLog_(data) {
  try {
    if (!data.gz) return jsonOut_({ ok: false, error: 'Nincs napló-tartalom (gz mező üres).' });

    const now = new Date();
    const ts = Utilities.formatDate(now, 'Europe/Budapest', 'yyyy-MM-dd_HH-mm-ss');
    const tsShort = Utilities.formatDate(now, 'Europe/Budapest', 'yyyy-MM-dd HH-mm');

    // Gyökér-mappa (a táblázat mellett) -> ezen belül FUTÁSONKÉNT egy almappa.
    const root = getLogFolder_();
    const folder = getRunFolder_(root, data.folder, tsShort);

    const gep = String(data.machine || 'gep').replace(/[^A-Za-z0-9_\-]/g, '_').substring(0, 40);
    const name = ts + '_' + gep + '_build' + (data.build || '') + '.log.gz';

    const blob = Utilities.newBlob(Utilities.base64Decode(data.gz), 'application/gzip', name);
    const file = folder.createFile(blob);
    const desc =
      'Eredmény: ' + (data.outcome || '-') +
      ' | Fájlok: ' + ((data.files || []).join(', ') || '-') +
      ' | Nyers méret: ' + (data.raw_bytes || 0) + ' bájt';
    file.setDescription(desc);
    // A leírás a MAPPÁRA is rákerül: a Drive listanézetében így a mappa megnyitása nélkül
    // látszik, mi volt a futás eredménye.
    try { if (folder.getId() !== root.getId()) folder.setDescription(desc); } catch (err2) {}

    cleanOldLogs_(root);

    return jsonOut_({ ok: true, url: file.getUrl(), name: name, folder: folder.getName() });
  } catch (err) {
    return jsonOut_({ ok: false, error: String(err) });
  }
}

/* FUTÁSONKÉNTI almappa a napló-gyökéren belül, pl.:
     "OTTHONI-PC - ASRock B450M Pro4 - 2026-08-27 16-45"
     "DESKTOP-8H3K2 - LENOVO ThinkPad T580 - 2026-08-27 17-20"

   A gépnév/típus a programtól jön (`folder` mező), a DÁTUMOT ITT fűzzük hozzá - egy óra,
   egy időzóna (Europe/Budapest), így a mappa dátuma és a benne lévő fájl időbélyege nem
   csúszhat el egymástól.

   Gépnév ELÖL: a Drive alapból név szerint rendez, így EGY gép összes futása egymás alatt
   sorakozik, a dátum meg időrendbe teszi őket.

   Ha a program nem küldött `folder` mezőt (régebbi build), a fájl a gyökérbe kerül -
   inkább menjen rossz helyre, mint hogy elvesszen. */
function getRunFolder_(root, base, tsShort) {
  let clean = String(base || '').replace(/[\/\\:*?"<>|]/g, '-').replace(/\s+/g, ' ').trim();
  if (!clean) return root;
  clean = clean.substring(0, 90) + ' - ' + tsShort;
  const it = root.getFoldersByName(clean);
  return it.hasNext() ? it.next() : root.createFolder(clean);
}

/* A napló-mappa megkeresése/létrehozása A TÁBLÁZAT MELLETT.

   Miért nem a `DriveApp.getFoldersByName(...)`: az az EGÉSZ Drive-ban keres név szerint,
   tehát ha bárhol máshol van már ilyen nevű mappa (akár egy megosztott meghajtón), a
   naplók oda kerülnének - és utána a technikus nem találná őket. A táblázat saját
   mappáján belül keresve ez nem fordulhat elő.

   Ha a táblázat a Saját meghajtó gyökerében van, a napló-mappa is oda kerül. */
function getLogFolder_() {
  let parent = null;
  try {
    const ssFile = DriveApp.getFileById(SpreadsheetApp.getActiveSpreadsheet().getId());
    const parents = ssFile.getParents();
    parent = parents.hasNext() ? parents.next() : null;
  } catch (err) {
    // Nem baj: lentebb a Drive-gyökér a tartalék. A naplót elveszíteni sokkal
    // rosszabb, mint rossz mappába tenni.
  }
  if (!parent) {
    try {
      parent = DriveApp.getRootFolder();
    } catch (err2) {
      // IDE AKKOR JUTUNK, HA A SCRIPTNEK EGYALTALAN NINCS DRIVE-ENGEDELYE.
      // A nyers "You do not have permission to call DriveApp.getRootFolder" ilyenkor
      // a PROGRAM kepernyojere is kikerulne, ahol semmit nem mond a technikusnak -
      // ezert cserejuk erre a mondatra, ami megnevezi a teendot. A 'auth/drive'
      // resz szandekos: a Python oldal (logupload_core.friendly_upload_error) erre
      // is illeszkedik, tehat a felulet tovabbra is a magyar teendo-szoveget mutatja.
      throw new Error('A szerviz Apps Script-jenek nincs Drive-engedelye (auth/drive). ' +
                      'Nyisd meg a scriptet, futtasd az "engedelyezes" fuggvenyt, es kovesd ' +
                      'a naplo ablakban megjeleno lepeseket, majd telepitsd uj verziokent.');
    }
  }
  const it = parent.getFoldersByName(LOG_FOLDER_NAME);
  return it.hasNext() ? it.next() : parent.createFolder(LOG_FOLDER_NAME);
}

function checkLogPassword_(pw) {
  return String(pw || '') === String(LOG_READ_PASSWORD) && String(LOG_READ_PASSWORD).length > 0;
}

/* A feltöltött naplók listája: futásonként egy sor. A tartalmat NEM küldi, csak a
   fejléc-adatokat - így a program megmutathatja, mi van fent, és csak azt tölti le,
   ami még nincs meg helyben. */
function listDriverLogs_() {
  try {
    const root = getLogFolder_();
    const rows = [];
    // Futás-almappák.
    const subs = root.getFolders();
    while (subs.hasNext() && rows.length < LOG_LIST_MAX * 2) {
      const d = subs.next();
      const files = d.getFiles();
      while (files.hasNext()) {
        rows.push(logRow_(files.next(), d.getName()));
      }
    }
    // A gyökérben maradt régi (almappa nélküli) naplók is látszódjanak.
    const loose = root.getFiles();
    while (loose.hasNext()) {
      const f = loose.next();
      if (f.getName().indexOf('.log.gz') >= 0) { rows.push(logRow_(f, '')); }
    }
    rows.sort((a, b) => (a.date < b.date ? 1 : (a.date > b.date ? -1 : 0)));
    return jsonOut_({ ok: true, logs: rows.slice(0, LOG_LIST_MAX), total: rows.length });
  } catch (err) {
    return jsonOut_({ ok: false, error: String(err) });
  }
}

function logRow_(file, folderName) {
  return {
    id: file.getId(),
    name: file.getName(),
    folder: folderName,
    size: file.getSize(),
    date: Utilities.formatDate(file.getDateCreated(), 'Europe/Budapest', 'yyyy-MM-dd HH:mm:ss'),
    desc: file.getDescription() || ''
  };
}

/* EGY napló tartalma (ugyanabban a gzip+base64 formában, ahogy feltöltötték).

   A `belongsToLogFolder_` ellenőrzés NEM formalitás: enélkül a végpont bárkinek, aki a
   jelszót ismeri, TETSZŐLEGES fájlt kiadna a Drive-ról, elég hozzá a fájl azonosítója.
   Így viszont csak a saját napló-mappájából szolgál ki. */
function getDriverLog_(data) {
  try {
    const id = String(data.id || '');
    if (!id) return jsonOut_({ ok: false, error: 'Nincs megadva fájl-azonosító.' });
    const file = DriveApp.getFileById(id);
    if (!belongsToLogFolder_(file)) {
      return jsonOut_({ ok: false, error: 'Ez a fájl nem a napló-mappában van.' });
    }
    return jsonOut_({
      ok: true,
      name: file.getName(),
      size: file.getSize(),
      gz: Utilities.base64Encode(file.getBlob().getBytes())
    });
  } catch (err) {
    return jsonOut_({ ok: false, error: String(err) });
  }
}

function belongsToLogFolder_(file) {
  const rootId = getLogFolder_().getId();
  const parents = file.getParents();
  while (parents.hasNext()) {
    const p = parents.next();
    if (p.getId() === rootId) return true;          // közvetlenül a gyökérben
    const gps = p.getParents();
    while (gps.hasNext()) {
      if (gps.next().getId() === rootId) return true;  // futás-almappában
    }
  }
  return false;
}

/* Régi naplók takarítása. Alapból KI van kapcsolva (LOG_RETENTION_DAYS = 0):
   naplót törölni visszafordíthatatlan, és pont a régi futások a leghasznosabbak,
   amikor egy visszatérő hibát keresel. Csak a saját gyökér-mappáját nézi, semmi mást:
   a futás-almappákat egyben dobja a kukába, plusz a gyökérben esetleg ott maradt régi
   (almappa nélküli) naplófájlokat. */
function cleanOldLogs_(root) {
  if (!LOG_RETENTION_DAYS || LOG_RETENTION_DAYS <= 0) return;
  const limit = new Date().getTime() - LOG_RETENTION_DAYS * 24 * 60 * 60 * 1000;
  const subs = root.getFolders();
  while (subs.hasNext()) {
    const d = subs.next();
    if (d.getDateCreated().getTime() < limit) { d.setTrashed(true); }
  }
  const files = root.getFiles();
  while (files.hasNext()) {
    const f = files.next();
    if (f.getName().indexOf('.log.gz') >= 0 && f.getDateCreated().getTime() < limit) {
      f.setTrashed(true);
    }
  }
}
